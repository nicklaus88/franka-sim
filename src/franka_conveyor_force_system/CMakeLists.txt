cmake_minimum_required(VERSION 3.16)
project(franka_conveyor_force_system)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport REQUIRED)
add_library(${PROJECT_NAME} SHARED src/ConveyorForce.cc)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(${PROJECT_NAME} PRIVATE gz-sim::gz-sim gz-plugin::register gz-transport::gz-transport)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_environment_hooks(env-hooks/conveyor_force.dsv.in)
ament_package()
