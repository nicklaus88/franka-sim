cmake_minimum_required(VERSION 3.16)
project(franka_tracking_controller)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/Velocity.cpp)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)
ament_target_dependencies(${PROJECT_NAME} controller_interface pluginlib rclcpp std_msgs)
pluginlib_export_plugin_description_file(controller_interface plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_package()
